# Dependencias de la aplicación.
#
# Python 3.10 de 64 bits: el wheel de stapipy que trae el SDK de Sentech es cp310,
# así que la versión del intérprete no es libre.
#
#   Windows:  .venv\Scripts\python.exe -m pip install -r requirements.txt
#   Linux:    .venv/bin/python -m pip install -r requirements.txt

# Núcleo
# numpy queda por debajo de 2.0: es lo que piden las librerías de deep learning
# (Torch, stardist, csbdeep) y los wheels compilados contra la ABI de numpy 1.x.
numpy>=1.24,<2.0
PyYAML>=6.0

# GUI y hilos de Qt: los QThread de system/ dependen de esto aunque no haya
# ventana, porque las señales son el cableado entre subsistemas.
PySide6>=6.5,<7

# Métricas de hardware (CPU, RAM, disco, contadores de red) para system_monitor.
# Multiplataforma: la misma API responde en Windows y en el aarch64 de la Jetson.
psutil>=5.9

# OpenCV de la misma época que numpy 1.x: los wheels 4.9+ se compilan contra la ABI
# de numpy 2 y no cargan con numpy 1.
#
# Variante con GUI a propósito, no la headless que usan los proyectos en producción:
# manual_test/camera_live_view.py muestra el feed con cv2.imshow. Cuando entre la UI
# de PySide6 y el visor deje de depender de imshow, pasar a opencv-python-headless
# (las dos instaladas a la vez se pisan).
opencv-python>=4.8,<4.9

# Cámaras Basler. El wheel incluye su propio runtime de pylon, así que alcanza para
# capturar: el SDK completo (viewer, IP Configurator, filter driver) es opcional y lo
# cubre setup/cameras/.
pypylon>=2.1.0

# Cámaras Sentech / Omron: stapipy NO está en PyPI. Lo instala el script de setup
# desde el wheel que hay que copiar a packages/:
#   Windows: setup\cameras\windows\sentech.ps1
#   Linux:   setup/cameras/linux/sentech.sh
# Se deja asentado acá porque es una dependencia real de la app, aunque pip no la
# pueda resolver.

# Servidor Modbus esclavo (system/modbus). Por debajo de la 3.8 el contexto del
# servidor se arma con `slaves=` en vez de `devices=`, y en la 4.0 desaparece
# ModbusSimulatorContext a favor de SimData/SimDevice: las dos puntas del rango
# están fijadas por la API que usa system/modbus/server.py.
pymodbus>=3.8,<4

# pymodbus no declara ninguna dependencia dura: el soporte de puerto serie es un
# extra. Sin pyserial el servidor Modbus RTU falla al arrancar —el TCP no lo
# necesita— y el motivo que se lee en el log es la librería faltante, no el puerto.
pyserial>=3.5

# Backends de telemetría (system/telemetry/backends). Los dos son opcionales: sin la
# librería instalada el backend degrada a no-op y lo avisa por `status`, así que
# se pueden sacar de acá en una instalación que no publique hacia afuera.
influxdb-client>=1.36
paho-mqtt>=1.6,<3

# Firma de las licencias (system/license). Ed25519: la clave pública son 32 bytes que
# entran en una línea del código y no hay parámetros que elegir mal. Hay wheels cp310
# para win_amd64 y para el aarch64 de la Jetson, así que no pelea con el pin de Python.
#
# A diferencia de los backends de telemetría, sin esta librería el módulo NO degrada a
# no-op: un verificador que no puede verificar rechaza todo. Un candado que no cierra
# tiene que quedar cerrado.
cryptography>=42.0

# Los tests y el lint están en requirements-dev.txt, que incluye este archivo.
